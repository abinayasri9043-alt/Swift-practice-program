1.Sum of digits

var number = 1234
var sum = 0

while number > 0 {
    sum += number % 10
    number /= 10
}
print("Sum of Digits:", sum)


2.Reverse a number

var number = 1234
var reverse = 0

while number > 0 {
    let digit = number % 10
    reverse = reverse * 10 + digit
    number /= 10
}
print("Reverse:", reverse)

3.Palindrome number

let number = 121
var temp = number
var reverse = 0

while temp > 0 {
    let digit = temp % 10
    reverse = reverse * 10 + digit
    temp /= 10
}
if number == reverse {
    print("Palindrome Number")
} else {
    print("Not a Palindrome")
}

4.Fibonacci series

var first = 0
var second = 1
for _ in 1...10 {
    print(first)

    let next = first + second
    first = second
    second = next
}
